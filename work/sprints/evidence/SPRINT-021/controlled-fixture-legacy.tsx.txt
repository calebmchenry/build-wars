import React,{useReducer,useState} from 'react';import {createRoot} from 'react-dom/client';
import {requireReadyCatalogs} from '../../../../src/app/catalogs';
import {BuildSetNavigator} from '../../../../src/app/components/BuildSetNavigator';
import {BuildSetTransferDialog} from '../../../../src/app/components/BuildSetTransferDialog';
import {PartyTransferDialog} from '../../../../src/app/components/PartyTransferDialog';
import {BuildComposer} from '../../../../src/app/components/BuildComposer';
import {selectValidationView} from '../../../../src/app/editor-selectors';
import {workspaceReducer,createInitialWorkspaceState,createWorkspaceEnvelope,workspacePersistenceFingerprint,materializeActiveDocument} from '../../../../src/app/workspace-state';
import {validBuildSetSavedRecordFixture,validPartyBuildSetSnapshotFixture,validLocalLibraryEnvelopeFixture,fixtureCatalogFacts} from '../../../../src/app/library-fixtures';
import {readLocalLibrary,writeLocalLibrary} from '../../../../src/app/local-storage';
import '../../../../src/app/styles.css';
const catalogs=requireReadyCatalogs();
function initial(){const stored=readLocalLibrary();if(stored.status!=='empty')return createInitialWorkspaceState({envelope:stored.envelope,writeBlocked:stored.writeBlocked,diagnostics:stored.diagnostics});const record=validBuildSetSavedRecordFixture({snapshot:validPartyBuildSetSnapshotFixture()});return createInitialWorkspaceState({envelope:validLocalLibraryEnvelopeFixture({revision:0,savedDocuments:[record],workingDraft:{document:record.document,associatedRecordId:record.id,savedWith:fixtureCatalogFacts}})});}
function Legacy(){const [workspace,dispatch]=useReducer(workspaceReducer,null,initial);const [transfer,setTransfer]=useState('');const [status,setStatus]=useState('Initial controlled legacy fixture');const validation=selectValidationView(workspace.editor,catalogs);return <main className="app-shell"><h1>Legacy workflow fixture — existing production components</h1><p>This isolated origin mounts retained build-set/party controls for regression verification. The main composer does not expose this test route.</p><button onClick={()=>{const now=new Date().toISOString();const result=writeLocalLibrary(window.localStorage,createWorkspaceEnvelope(workspace,fixtureCatalogFacts,now),{now,reason:"legacy-browser-check",expectedRevision:workspace.storage.revision});dispatch({type:'storage-write-result',result,durableFingerprint:workspacePersistenceFingerprint(workspace,fixtureCatalogFacts)});setStatus(result.ok?'Fixture library saved':'Fixture library save failed: '+result.message);}}>Save fixture library</button><p role="status">{status}</p><BuildSetNavigator workspace={workspace} catalogs={catalogs} dispatch={dispatch} onOpenTransfer={()=>setTransfer('set')} onOpenPartyTransfer={()=>setTransfer('party')}/><BuildComposer workspace={workspace} catalogs={catalogs} validation={validation} editorDispatch={action=>dispatch({type:'editor',action})} workspaceDispatch={dispatch} requestDraftReplacement={()=>'discard'}/><details><summary>Verification snapshot</summary><textarea aria-label="Verification snapshot" readOnly value={JSON.stringify(materializeActiveDocument(workspace))}/></details><BuildSetTransferDialog open={transfer==='set'} workspace={workspace} dispatch={dispatch} onClose={()=>setTransfer('')}/><PartyTransferDialog open={transfer==='party'} workspace={workspace} dispatch={dispatch} onClose={()=>setTransfer('')}/></main>;}
createRoot(document.getElementById('root')!).render(<Legacy/>);
